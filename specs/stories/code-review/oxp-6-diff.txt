+export function manoeuvreDotReasons(rank, count, left) {
+  const r = Math.max(0, Math.min(count, Math.trunc(rank) || 0));
+  const known = Number.isFinite(left);
+  const reasons = [];

+    }

+  }
+  return reasons;
