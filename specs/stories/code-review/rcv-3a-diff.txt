+ * function's own documented contract.

+ * vitae_cost/willpower_cost: shared/resist.js builds synthetic `pi` objects for

+  if (!box) return;

+  const hasRules = pi && (pi.effect || pi.action || pi.duration || pi.cost);
+  if (!hasRules) {
+    box.style.display = 'none';
+    box.open = false;
+    if (bodyEl) bodyEl.innerHTML = '';
+    return;
+  }
